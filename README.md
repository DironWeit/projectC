#### В файле logic.c
```c

int my_getch() {
// Скопирую старые настройки терминала и скажу, что будут использоваться новые
struct termios oldt, newt;
// создал переменную для символа
int ch;
// получаю текущие настройки STDIN_FILENO - стандартный ввод (обычно клавиатура)
tcgetattr(STDIN_FILENO, &oldt);
// копирую старые настройки в новые
newt = oldt;
// меняю флаги в настройке (подробней в сноске #1)
newt.c_lflag &= ~(ICANON | ECHO);
// применяю новые настройки
tcsetattr(STDIN_FILENO, TCSANOW, &newt);
// Читаю символ
ch = getchar();
// Возвращаю старые настройки
tcsetattr(STDIN_FILENO, TCSANOW, &oldt);
// возвращаю символ
return ch;
}

```

Сноска №1:
Разберем эту строку: ```newt.c_lflag &= ~(ICANON | ECHO);```  
c_lflag - локальные флаги  
| Режим | Флаг | Обычное состояние (ВКЛ) | Наше состояние (ВЫКЛ) |
|-------|------|------------------------|----------------------|
| **Канонический режим** | `ICANON` | Терминал буферизирует ввод, отправляет программе только после нажатия Enter | Каждый символ сразу поступает в программу |
| **Эхо** | `ECHO` | Нажатые клавиши отображаются на экране | Символы НЕ показываются на экране |

Операция &= ~:  
~ - побитовое НЕ (инвертирует биты)  
&= - побитовое И с присваиванием

```c
void draw_tile(int value, int x, int y) { 
  SDL_Rect rect;

  // создается прямоугольник
  rect.x = x;
  rect.y = y;
  rect.w = TILE_SIZE - 10;
  rect.h = TILE_SIZE - 10;

  // Цвет плитки 
  if(value == 0) {
    SDL_SetRenderDrawColor(renderer,180,180,180,255); // пустая клетка - серая 
  } else {
    SDL_SetRenderDrawColor(renderer,255,180,0,255); // заполненная клетка - оранживая
  }

  // рисуем прямоугольник
  SDL_RenderFillRect(renderer,&rect);
}
```

```c
void gui_draw(int pole[4][4]){
  //задаем цвет фона и очищаем экран
  SDL_SetRenderDrawColor(renderer,220,220,220,255);
  SDL_RenderClear(renderer);
  // идем по все матрице
  for(int i=0;i<GRID_SIZE;i++){
    for(int j=0;j<GRID_SIZE;j++){
      // вычисляем координаты
      int x = j*TILE_SIZE + 50;
      int y = i*TILE_SIZE + 50;
      // рисуем плитку  
      draw_tile(pole[i][j],x,y);
    }
  }
  // это рендерит все

}
```

```c
// обработка ввода
int gui_handle_input(int pole[4][4]) {
  SDL_Event event;
  
  while(SDL_PollEvent(&event)) { // чтение почереди
    if(event.type == SDL_QUIT) // обработка нажатия
    return 0;
    if(event.type == SDL_KEYDOWN) {
      switch(event.key.keysym.sym) // управление стрелками
      {
        case SDLK_UP: // верх
        moveTop(pole);
        break;

        case SDLK_DOWN:// низ
        moveBottom(pole);
        break;
        
        case SDLK_LEFT: // лево
        moveLeft(pole);
        break;
        
        case SDLK_RIGHT: //право
        moveRight(pole);
        break;
      }
    }
  }
  
  return 1;
}
```
